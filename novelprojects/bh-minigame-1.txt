    import pygame
    import renpy.exports as renpy
    import random

    # Относительные размеры (проценты от экрана)
    SCREEN_PADDING = 0.02
    GAME_ZONE_WIDTH_RATIO = 0.60
    SIDE_PANEL_WIDTH_RATIO = 0.20
    TOP_PANEL_HEIGHT_RATIO = 0.10
    BOTTOM_PANEL_HEIGHT_RATIO = 0.06

    # Время игры в секундах
    GAME_DURATION = 120.0  # 2 минуты
    CRITICAL_TIME = 46   

    # Типы атак
    ATTACK_BASIC = 1        # Базовые атаки
    ATTACK_CIRCLE = 2       # Атаки по кругу
    ATTACK_SNIPER = 3       # Сверхскоростные снаряды

    # Расчет абсолютных размеров
    def calculate_sizes():
        screen_width = renpy.config.screen_width
        screen_height = renpy.config.screen_height

        return {
            "screen_width": screen_width,
            "screen_height": screen_height,
            "padding": int(screen_width * SCREEN_PADDING),
            "side_panel_width": int(screen_width * SIDE_PANEL_WIDTH_RATIO),
            "top_panel_height": int(screen_height * TOP_PANEL_HEIGHT_RATIO),
            "bottom_panel_height": int(screen_height * BOTTOM_PANEL_HEIGHT_RATIO),
            "game_zone_width": int(screen_width * GAME_ZONE_WIDTH_RATIO),
        }

    # Глобальные переменные с размерами
    sizes = None

    class Assistant(renpy.Displayable):
        def __init__(self, *args, **kwargs):
            super(Assistant, self).__init__(*args, **kwargs)

            global sizes
            if sizes is None:
                sizes = calculate_sizes()

            # Размеры помощника
            self.radius = int(renpy.config.screen_height * 0.04)
            self.base_color = "#87CEEB"  # Голубой цвет
            self.active_color = "#00BFFF"  # Яркий голубой
            self.current_color = self.base_color

            # Позиция в правом нижнем углу
            self.x = sizes["screen_width"] - sizes["padding"] - self.radius * 2
            self.y = sizes["screen_height"] - sizes["bottom_panel_height"] - self.radius * 2

            # Способности
            self.ability_cooldowns = [0, 0, 0]
            self.ability_durations = [0, 0, 0]
            self.ability_messages = ["", "", ""]
            self.last_ability_time = 0
            self.min_time_between_abilities = 3.0

            # Анимации
            self.pulse_phase = 0
            self.pulse_speed = 2.0
            self.is_active = False
            self.active_end_time = 0

            self.old_st = None

        def update(self, dtime, current_time, player_lives, max_lives, player_took_hit):
            # Обновляем откаты способностей
            for i in range(3):
                if self.ability_cooldowns[i] > 0:
                    self.ability_cooldowns[i] = max(0, self.ability_cooldowns[i] - dtime)

            # Обновляем длительности активных способностей
            for i in range(3):
                if self.ability_durations[i] > 0:
                    self.ability_durations[i] = max(0, self.ability_durations[i] - dtime)
                    if self.ability_durations[i] <= 0:
                        self.ability_messages[i] = ""

            # Проверяем, можно ли использовать способности
            can_use_ability = (current_time - self.last_ability_time >= self.min_time_between_abilities)

            # Определяем приоритет способностей
            ability_used = False

            # Способность 1: Восстановление здоровья (активируется чаще)
            if (not ability_used and can_use_ability and
                player_lives < 3 and self.ability_cooldowns[0] <= 0):
                self.activate_ability(0, current_time, "Ты справишься!")
                ability_used = True
                return 1

            # Способность 2: Щит неуязвимости (активируется при получении урона)
            elif (not ability_used and can_use_ability and
                  player_took_hit and self.ability_cooldowns[1] <= 0):
                self.activate_ability(1, current_time, "Ты выдержишь!")
                ability_used = True
                return 2

            # Способность 3: Замедление времени (активируется при 1 HP)
            elif (not ability_used and can_use_ability and
                  player_lives == 1 and self.ability_cooldowns[2] <= 0):
                self.activate_ability(2, current_time, "Ты не проиграешь!")
                ability_used = True
                return 3

            return 0

        def activate_ability(self, ability_index, current_time, message):
            self.ability_cooldowns[ability_index] = 15.0  # Уменьшено с 20 до 15 секунд
            self.ability_durations[ability_index] = 3.0
            self.ability_messages[ability_index] = message
            self.last_ability_time = current_time
            self.is_active = True
            self.active_end_time = current_time + 1.0

        def render(self, width, height, st, at):
            render = renpy.Render(width, height)

            if self.old_st is None:
                self.old_st = st

            dtime = st - self.old_st
            self.old_st = st

            current_time = renpy.get_game_runtime()

            # Обновляем пульсацию
            self.pulse_phase += self.pulse_speed * dtime

            # Проверяем активное свечение
            if self.is_active and current_time >= self.active_end_time:
                self.is_active = False

            # Выбираем цвет
            if self.is_active:
                pulse_factor = (math.sin(self.pulse_phase) + 1) / 2
                r1, g1, b1 = int(self.base_color[1:3], 16), int(self.base_color[3:5], 16), int(self.base_color[5:7], 16)
                r2, g2, b2 = int(self.active_color[1:3], 16), int(self.active_color[3:5], 16), int(self.active_color[5:7], 16)

                r = int(r1 + (r2 - r1) * pulse_factor)
                g = int(g1 + (g2 - g1) * pulse_factor)
                b = int(b1 + (b2 - b1) * pulse_factor)

                self.current_color = "#{:02X}{:02X}{:02X}".format(r, g, b)
            else:
                self.current_color = self.base_color

            # Рисуем помощника
            circle_surface = renpy.display.pgrender.surface((self.radius * 2, self.radius * 2), True)
            import pygame as pg
            pg.draw.circle(circle_surface, renpy.easy.color(self.current_color),
                          (self.radius, self.radius), self.radius)

            # Обводка
            pg.draw.circle(circle_surface, renpy.easy.color("#000000"),
                          (self.radius, self.radius), self.radius, 2)

            render.blit(circle_surface, (self.x, self.y))

            renpy.redraw(self, 0.01)
            return render

        def event(self, ev, x, y, st):
            raise renpy.IgnoreEvent()

    class BulletHellPlayer(renpy.Displayable):
        def __init__(self, color="#ff0000", size_ratio=0.025, speed=(0, 0), *args, **kwargs):
            super(BulletHellPlayer, self).__init__(*args, **kwargs)

            self.color = color
            self.size = int(renpy.config.screen_height * size_ratio)
            self.displayable = Solid(color, xsize=self.size, ysize=self.size)

            # Основное движение
            self.speed_x = 0
            self.speed_y = 0
            # Увеличена скорость игрока для лучшей маневренности
            self.max_speed_x = speed[0] * renpy.config.screen_width / 1280 * 0.9
            self.max_speed_y = speed[1] * renpy.config.screen_height / 720 * 0.9

            # Рывок
            self.dash_distance = self.size * 6
            self.dash_cooldown = 0.4  # Уменьшено с 0.5
            self.dash_duration = 0.15
            self.is_dashing = False
            self.dash_start_time = 0
            self.dash_direction_x = 0
            self.dash_direction_y = 0
            self.dash_cooldown_timer = 0
            self.dash_ready = True
            self.dash_from_x = 0
            self.dash_from_y = 0
            self.last_move_x = 0
            self.last_move_y = 0

            global sizes
            if sizes is None:
                sizes = calculate_sizes()

            game_zone_x = sizes["padding"] + sizes["side_panel_width"]
            game_zone_y = sizes["top_panel_height"]
            game_zone_width = sizes["game_zone_width"]
            game_zone_height = sizes["screen_height"] - sizes["top_panel_height"] - sizes["bottom_panel_height"]

            self.x = game_zone_x + game_zone_width // 2 - self.size // 2
            self.y = game_zone_y + game_zone_height // 2 - self.size // 2

            self.old_st = None
            self.alive = True
            self.displayable_render = None
            self.max_lives = 3
            self.lives = self.max_lives
            self.last_hit_time = -10
            self.invincible_time = 1.2  # Увеличено с 1.0
            self.invincible_flash_phase = 0.0
            self.is_invincible = False
            self.just_took_hit = False
            self.last_lives = self.max_lives

            # Эффекты от помощника
            self.shield_active = False
            self.shield_end_time = 0
            self.time_slow_active = False
            self.time_slow_end_time = 0
            self.time_slow_radius = self.size * 10  # Увеличен радиус
            self.heal_animation_time = 0
            self.lost_life_flash_time = 0

        @property
        def width(self):
            return self.size

        @property
        def height(self):
            return self.size

        def overlaps_with(self, other):
            dx = abs((self.x + self.size/2) - (other.x + other.size/2))
            dy = abs((self.y + self.size/2) - (other.y + other.size/2))

            # Увеличена зона столкновения для прощения
            return dx < (self.size + other.size)/2.5 and dy < (self.size + other.size)/2.5

        def move_player(self):
            current_time = renpy.get_game_runtime()

            # Обновляем откат рывка
            if not self.dash_ready:
                self.dash_cooldown_timer = current_time - self.dash_start_time
                if self.dash_cooldown_timer >= self.dash_cooldown:
                    self.dash_ready = True

            # Если активен рывок
            if self.is_dashing:
                dash_progress = (current_time - self.dash_start_time) / self.dash_duration

                if dash_progress >= 1.0:
                    self.is_dashing = False
                    self.speed_x = 0
                    self.speed_y = 0
                else:
                    self.speed_x = 0
                    self.speed_y = 0
                    return

            # Обычное управление
            keys = pygame.key.get_pressed()

            self.last_move_x = 0
            self.last_move_y = 0

            if keys[pygame.K_w] or keys[pygame.K_UP]:
                self.speed_y = -self.max_speed_y
                self.last_move_y = -1
            if keys[pygame.K_s] or keys[pygame.K_DOWN]:
                self.speed_y = self.max_speed_y
                self.last_move_y = 1
            if keys[pygame.K_a] or keys[pygame.K_LEFT]:
                self.speed_x = -self.max_speed_x
                self.last_move_x = -1
            if keys[pygame.K_d] or keys[pygame.K_RIGHT]:
                self.speed_x = self.max_speed_x
                self.last_move_x = 1

            if not (keys[pygame.K_w] or keys[pygame.K_UP] or keys[pygame.K_s] or keys[pygame.K_DOWN]):
                self.speed_y = 0
            if not (keys[pygame.K_a] or keys[pygame.K_LEFT] or keys[pygame.K_d] or keys[pygame.K_RIGHT]):
                self.speed_x = 0

            # Активация рывка
            if (keys[pygame.K_SPACE] or keys[pygame.K_LSHIFT]) and self.dash_ready and not self.is_dashing:
                self.activate_dash()

        def activate_dash(self):
            if self.last_move_x == 0 and self.last_move_y == 0:
                self.dash_direction_x = 1
                self.dash_direction_y = 0
            else:
                length = math.sqrt(self.last_move_x**2 + self.last_move_y**2)
                self.dash_direction_x = self.last_move_x / length
                self.dash_direction_y = self.last_move_y / length

            self.dash_from_x = self.x
            self.dash_from_y = self.y

            self.is_dashing = True
            self.dash_ready = False
            self.dash_start_time = renpy.get_game_runtime()

            # Во время рывка неуязвим
            self.last_hit_time = self.dash_start_time

        def render(self, width, height, st, at):
            render = renpy.Render(width, height)

            if self.old_st is None:
                self.old_st = st

            dtime = st - self.old_st
            self.old_st = st

            current_time = renpy.get_game_runtime()
            time_since_hit = current_time - self.last_hit_time

            # Обновляем эффекты
            if self.shield_active and current_time >= self.shield_end_time:
                self.shield_active = False

            if self.time_slow_active and current_time >= self.time_slow_end_time:
                self.time_slow_active = False

            if self.heal_animation_time > 0:
                self.heal_animation_time = max(0, self.heal_animation_time - dtime)

            if self.lost_life_flash_time > 0:
                self.lost_life_flash_time = max(0, self.lost_life_flash_time - dtime)

            # Проверяем состояние неуязвимости
            self.is_invincible = (time_since_hit < self.invincible_time or self.shield_active)

            if self.is_invincible:
                self.invincible_flash_phase = (math.sin(current_time * 15) + 1) / 2  # Медленнее мигание
            else:
                self.invincible_flash_phase = 1.0

            if self.alive:
                self.move_player()

                if self.is_dashing:
                    dash_progress = (current_time - self.dash_start_time) / self.dash_duration

                    if dash_progress < 1.0:
                        progress = dash_progress
                        ease_progress = 1 - (1 - progress) ** 2

                        self.x = self.dash_from_x + self.dash_direction_x * self.dash_distance * ease_progress
                        self.y = self.dash_from_y + self.dash_direction_y * self.dash_distance * ease_progress
                else:
                    self.x += self.speed_x * dtime
                    self.y += self.speed_y * dtime

                global sizes
                game_zone_x = sizes["padding"] + sizes["side_panel_width"]
                game_zone_y = sizes["top_panel_height"]
                game_zone_width = sizes["game_zone_width"]
                game_zone_height = sizes["screen_height"] - sizes["top_panel_height"] - sizes["bottom_panel_height"]

                self.x = max(game_zone_x, min(self.x, game_zone_x + game_zone_width - self.size))
                self.y = max(game_zone_y, min(self.y, game_zone_y + game_zone_height - self.size))

            # Рисуем эффект замедления времени
            if self.time_slow_active:
                slow_circle = Circle(self.time_slow_radius, "#87CEEB80")
                slow_render = renpy.render(slow_circle, self.time_slow_radius * 2, self.time_slow_radius * 2, st, at)
                render.blit(slow_render, (self.x + self.size/2 - self.time_slow_radius,
                                         self.y + self.size/2 - self.time_slow_radius))

            # Рисуем эффект щита
            if self.shield_active:
                shield_radius = self.size * 1.5
                pulse = (math.sin(current_time * 8) + 1) / 2  # Медленнее пульсация
                shield_size = int(shield_radius * (0.8 + 0.2 * pulse))

                shield_circle = Circle(shield_size, "#FFD70080")
                shield_render = renpy.render(shield_circle, shield_size * 2, shield_size * 2, st, at)
                render.blit(shield_render, (self.x + self.size/2 - shield_size,
                                           self.y + self.size/2 - shield_size))

            # Рисуем игрока
            if self.is_invincible:
                # Более плавное мигание
                if int(current_time * 8) % 2 == 0:  # Медленнее мигание
                    self.displayable_render = renpy.render(
                        self.displayable,
                        width,
                        height,
                        st,
                        at
                    )

                    if self.alive:
                        render.blit(self.displayable_render, (self.x, self.y))
            else:
                self.displayable_render = renpy.render(
                    self.displayable,
                    width,
                    height,
                    st,
                    at
                )
                if self.alive:
                    render.blit(self.displayable_render, (self.x, self.y))

            renpy.redraw(self, 0.01)
            return render

        def take_hit(self, current_time):
            if not self.is_dashing and not self.is_invincible:
                self.lives -= 1
                self.last_hit_time = current_time
                self.just_took_hit = True
                self.lost_life_flash_time = 1.0
                return True
            return False

        def heal(self, amount=1):
            if self.lives < self.max_lives:
                self.lives = min(self.max_lives, self.lives + amount)
                self.heal_animation_time = 1.0
                return True
            return False

        def activate_shield(self, duration=3.0):
            current_time = renpy.get_game_runtime()
            self.shield_active = True
            self.shield_end_time = current_time + duration
            self.last_hit_time = current_time

        def activate_time_slow(self, duration=3.0):
            current_time = renpy.get_game_runtime()
            self.time_slow_active = True
            self.time_slow_end_time = current_time + duration

        def reset_hit_flag(self):
            self.just_took_hit = False

        def event(self, ev, x, y, st):
            if not self.alive:
                return True

            raise renpy.IgnoreEvent()

    class Circle(renpy.Displayable):
        def __init__(self, radius, color, *args, **kwargs):
            super(Circle, self).__init__(*args, **kwargs)
            self.radius = radius
            self.color = color

        def render(self, width, height, st, at):
            render = renpy.Render(self.radius * 2, self.radius * 2)

            circle_surface = renpy.display.pgrender.surface((self.radius * 2, self.radius * 2), True)

            import pygame as pg
            pg.draw.circle(circle_surface, renpy.easy.color(self.color),
                          (self.radius, self.radius), self.radius)

            render.blit(circle_surface, (0, 0))
            return render

    class TriangleBullet(renpy.Displayable):
        def __init__(self, speed=(0, 0), start=(0, 0), bullet_type=ATTACK_BASIC, *args, **kwargs):
            super(TriangleBullet, self).__init__(*args, **kwargs)

            # Уменьшены размеры пуль для прощения
            self.length = int(renpy.config.screen_height * 0.025)  # Уменьшено
            self.base = int(renpy.config.screen_height * 0.017)    # Уменьшено

            self.original_speed_x = speed[0] * renpy.config.screen_width / 1280 * 0.5  # Уменьшена скорость
            self.original_speed_y = speed[1] * renpy.config.screen_height / 720 * 0.5  # Уменьшена скорость
            self.speed_x = self.original_speed_x
            self.speed_y = self.original_speed_y

            self.x = start[0]
            self.y = start[1]

            self.bullet_type = bullet_type

            # Базовые параметры
            self.old_st = None
            self.alive = True
            self.hit_registered = False
            self.start_time = renpy.get_game_runtime()
            self.max_lifetime = 12.0  # Уменьшено время жизни

            if abs(self.original_speed_x) < 0.01 and abs(self.original_speed_y) < 0.01:
                self.angle = 0
            else:
                self.angle = math.atan2(self.original_speed_y, self.original_speed_x)

            self.slow_factor = 1.0

        @property
        def width(self):
            return max(self.length, self.base)

        @property
        def height(self):
            return max(self.length, self.base)

        def apply_time_slow(self, player_x, player_y, player_slow_radius, slow_factor=0.4):  # Сильнее замедление
            dx = self.x - player_x
            dy = self.y - player_y
            distance = math.sqrt(dx*dx + dy*dy)

            if distance < player_slow_radius:
                self.slow_factor = slow_factor
            else:
                self.slow_factor = 1.0

            self.speed_x = self.original_speed_x * self.slow_factor
            self.speed_y = self.original_speed_y * self.slow_factor

        def update_movement(self, dtime):
            self.x += self.speed_x * dtime
            self.y += self.speed_y * dtime

        def render(self, width, height, st, at):
            render = renpy.Render(width, height)

            if self.old_st is None:
                self.old_st = st

            dtime = st - self.old_st
            self.old_st = st

            current_time = renpy.get_game_runtime()

            # Обновляем движение
            self.update_movement(dtime)

            # Все пули белые с черной обводкой
            fill_color = "#FFFFFF"
            outline_color = "#000000"

            triangle_composite = Composite(
                (self.length, self.base),
                (0, 0), Solid(outline_color, xsize=self.length, ysize=self.base),
                (2, 2), Solid(fill_color, xsize=self.length-4, ysize=self.base-4)
            )

            triangle_transform = Transform(
                triangle_composite,
                rotate=math.degrees(self.angle),
                anchor=(0.5, 0.5)
            )

            triangle_render = renpy.render(triangle_transform, self.length, self.base, st, at)
            render.blit(triangle_render, (self.x - self.length//2, self.y - self.base//2))

            global sizes
            game_zone_x = sizes["padding"] + sizes["side_panel_width"]
            game_zone_y = sizes["top_panel_height"]
            game_zone_width = sizes["game_zone_width"]
            game_zone_height = sizes["screen_height"] - sizes["top_panel_height"] - sizes["bottom_panel_height"]

            margin = max(game_zone_width, game_zone_height) * 2.0

            if (self.y > game_zone_y + game_zone_height + margin or
                self.y < game_zone_y - margin or
                self.x > game_zone_x + game_zone_width + margin or
                self.x < game_zone_x - margin):
                self.alive = False

            if current_time - self.start_time > self.max_lifetime:
                self.alive = False

            renpy.redraw(self, 0)
            return render

    class BulletHellGame(renpy.Displayable):
        def __init__(self, *args, **kwargs):
            super(BulletHellGame, self).__init__(*args, **kwargs)

            global sizes
            sizes = calculate_sizes()

            self.player = BulletHellPlayer(
                color="#ff0000",
                size_ratio=0.025,
                speed=(400, 400)
            )

            self.assistant = Assistant()
            self.bullets = []

            # Таймеры для разных типов атак
            self.last_basic_attack_time = renpy.get_game_runtime()
            self.last_circle_attack_time = renpy.get_game_runtime()
            self.last_sniper_attack_time = renpy.get_game_runtime()

            # Интервалы атак (увеличены для прощения)
            self.basic_attack_interval = 1.2  # Увеличено
            self.circle_attack_interval = 10.0  # Увеличено
            self.sniper_attack_interval = 8.0   # Увеличено

            self.game_over = False
            self.game_over_time = None
            self.time_elapsed = 0.0
            self.victory = False
            self.game_start_time = renpy.get_game_runtime()
            self.critical_time_reached = False
            self.current_difficulty = 1.0
            self.music_started = False

            # Для сообщений помощника
            self.assistant_message = ""
            self.message_end_time = 0

        def add_basic_attack(self):
            global sizes
            game_zone_x = sizes["padding"] + sizes["side_panel_width"]
            game_zone_y = sizes["top_panel_height"]
            game_zone_width = sizes["game_zone_width"]
            game_zone_height = sizes["screen_height"] - sizes["top_panel_height"] - sizes["bottom_panel_height"]

            side = random.choice(["top", "bottom", "left", "right"])

            spawn_distance = int(max(game_zone_width, game_zone_height) * 0.5)

            if side == "top":
                start_x = random.randint(game_zone_x - spawn_distance,
                                        game_zone_x + game_zone_width + spawn_distance)
                start_y = game_zone_y - spawn_distance
                target_x = game_zone_x + game_zone_width - (start_x - game_zone_x)
                target_y = game_zone_y + game_zone_height + spawn_distance
            elif side == "bottom":
                start_x = random.randint(game_zone_x - spawn_distance,
                                        game_zone_x + game_zone_width + spawn_distance)
                start_y = game_zone_y + game_zone_height + spawn_distance
                target_x = game_zone_x + game_zone_width - (start_x - game_zone_x)
                target_y = game_zone_y - spawn_distance
            elif side == "left":
                start_x = game_zone_x - spawn_distance
                start_y = random.randint(game_zone_y - spawn_distance,
                                        game_zone_y + game_zone_height + spawn_distance)
                target_x = game_zone_x + game_zone_width + spawn_distance
                target_y = game_zone_y + game_zone_height - (start_y - game_zone_y)
            else:
                start_x = game_zone_x + game_zone_width + spawn_distance
                start_y = random.randint(game_zone_y - spawn_distance,
                                        game_zone_y + game_zone_height + spawn_distance)
                target_x = game_zone_x - spawn_distance
                target_y = game_zone_y + game_zone_height - (start_y - game_zone_y)

            dx = target_x - start_x
            dy = target_y - start_y

            distance = math.sqrt(dx*dx + dy*dy)
            if distance > 0:
                base_speed = 150 * self.current_difficulty  # Уменьшена базовая скорость
                speed_x = (dx / distance) * base_speed
                speed_y = (dy / distance) * base_speed
            else:
                speed_x = 0
                speed_y = 0

            # Создаем 1-2 пули вместо 1
            num_bullets = random.randint(1, 2)
            for _ in range(num_bullets):
                # Немного рандомизируем направление
                angle_variation = random.uniform(-0.1, 0.1)
                speed_x_varied = speed_x * math.cos(angle_variation) - speed_y * math.sin(angle_variation)
                speed_y_varied = speed_x * math.sin(angle_variation) + speed_y * math.cos(angle_variation)

                bullet = TriangleBullet(
                    speed=(speed_x_varied, speed_y_varied),
                    start=(start_x, start_y),
                    bullet_type=ATTACK_BASIC
                )
                self.bullets.append(bullet)

        def add_circle_attack(self):
            global sizes
            game_zone_x = sizes["padding"] + sizes["side_panel_width"]
            game_zone_y = sizes["top_panel_height"]
            game_zone_width = sizes["game_zone_width"]
            game_zone_height = sizes["screen_height"] - sizes["top_panel_height"] - sizes["bottom_panel_height"]

            spawn_distance = int(max(game_zone_width, game_zone_height) * 0.3)

            # Создаем снаряды по кругу (каждые 30 градусов) - только 8 пуль
            num_bullets = 8  # Уменьшено с 12 до 8
            base_speed = 120 * self.current_difficulty  # Уменьшена скорость

            for i in range(num_bullets):
                angle = math.radians(i * 45)  # 45 градусов между пулями (вместо 30)

                # Стартовая позиция на окружности вокруг арены
                start_x = game_zone_x + game_zone_width // 2 + math.cos(angle) * (game_zone_width // 2 + spawn_distance)
                start_y = game_zone_y + game_zone_height // 2 + math.sin(angle) * (game_zone_height // 2 + spawn_distance)

                # Направление к центру
                speed_x = -math.cos(angle) * base_speed
                speed_y = -math.sin(angle) * base_speed

                bullet = TriangleBullet(
                    speed=(speed_x, speed_y),
                    start=(start_x, start_y),
                    bullet_type=ATTACK_CIRCLE
                )
                self.bullets.append(bullet)

        def add_sniper_attack(self):
            global sizes
            game_zone_x = sizes["padding"] + sizes["side_panel_width"]
            game_zone_y = sizes["top_panel_height"]
            game_zone_width = sizes["game_zone_width"]
            game_zone_height = sizes["screen_height"] - sizes["top_panel_height"] - sizes["bottom_panel_height"]

            # Определяем противоположные стороны
            side_pairs = [
                ("top", "bottom"),
                ("bottom", "top"),
                ("left", "right"),
                ("right", "left")
            ]

            side1, side2 = random.choice(side_pairs)

            spawn_distance = int(max(game_zone_width, game_zone_height) * 0.5)

            # Количество пуль в серии - сильно уменьшено
            num_bullets = random.randint(3, 5)  # 3-5 пуль в обычном режиме
            if self.time_elapsed > CRITICAL_TIME:
                num_bullets = random.randint(4, 6)  # 4-6 пуль в критическом режиме

            # Создаем все пули из одной точки
            if side1 == "top":
                point1_x = random.randint(game_zone_x, game_zone_x + game_zone_width)
                point1_y = game_zone_y - spawn_distance
            elif side1 == "bottom":
                point1_x = random.randint(game_zone_x, game_zone_x + game_zone_width)
                point1_y = game_zone_y + game_zone_height + spawn_distance
            elif side1 == "left":
                point1_x = game_zone_x - spawn_distance
                point1_y = random.randint(game_zone_y, game_zone_y + game_zone_height)
            else:  # right
                point1_x = game_zone_x + game_zone_width + spawn_distance
                point1_y = random.randint(game_zone_y, game_zone_y + game_zone_height)

            for _ in range(num_bullets):
                # Вторая точка на противоположной стороне (немного рандомизирована)
                if side2 == "top":
                    point2_x = random.randint(game_zone_x, game_zone_x + game_zone_width)
                    point2_y = game_zone_y - spawn_distance
                elif side2 == "bottom":
                    point2_x = random.randint(game_zone_x, game_zone_x + game_zone_width)
                    point2_y = game_zone_y + game_zone_height + spawn_distance
                elif side2 == "left":
                    point2_x = game_zone_x - spawn_distance
                    point2_y = random.randint(game_zone_y, game_zone_y + game_zone_height)
                else:  # right
                    point2_x = game_zone_x + game_zone_width + spawn_distance
                    point2_y = random.randint(game_zone_y, game_zone_y + game_zone_height)

                # Задержка для создания веера
                import time
                renpy.invoke_in_thread(self.add_delayed_sniper_bullet, point1_x, point1_y, point2_x, point2_y, random.uniform(0, 0.3))

        def add_delayed_sniper_bullet(self, start_x, start_y, target_x, target_y, delay):
            import time
            time.sleep(delay)

            dx = target_x - start_x
            dy = target_y - start_y

            distance = math.sqrt(dx*dx + dy*dy)
            if distance > 0:
                # Сверхскоростной снаряд
                base_speed = 400 * self.current_difficulty  # Уменьшена скорость
                speed_x = (dx / distance) * base_speed
                speed_y = (dy / distance) * base_speed
            else:
                speed_x = 0
                speed_y = 0

            bullet = TriangleBullet(
                speed=(speed_x, speed_y),
                start=(start_x, start_y),
                bullet_type=ATTACK_SNIPER
            )
            self.bullets.append(bullet)

        def check_collisions(self):
            current_time = renpy.get_game_runtime()

            for bullet in self.bullets:
                if bullet.alive and self.player.alive and not bullet.hit_registered:
                    player_center_x = self.player.x + self.player.size/2
                    player_center_y = self.player.y + self.player.size/2
                    bullet_center_x = bullet.x
                    bullet_center_y = bullet.y

                    distance = ((player_center_x - bullet_center_x)**2 + (player_center_y - bullet_center_y)**2)**0.5

                    if distance < (self.player.size + bullet.length)/2:
                        if self.player.take_hit(current_time):
                            bullet.alive = False
                            bullet.hit_registered = True

                            if self.player.lives <= 0:
                                self.player.alive = False
                                self.game_over = True
                                self.game_over_time = current_time
                                self.victory = False
                            return True
            return False

        def update_difficulty(self):
            current_time = renpy.get_game_runtime()
            self.time_elapsed = current_time - self.game_start_time

            if not self.music_started and self.time_elapsed > 0.1:
                renpy.music.play("audio/heaven.mp3", channel="music", loop=True, fadein=1.0)
                self.music_started = True

            time_factor = min(1.8, 1.0 + self.time_elapsed / 80.0)  # Медленнее рост

            if self.time_elapsed > CRITICAL_TIME and not self.critical_time_reached:
                self.critical_time_reached = True
                self.current_difficulty = 1.8  # Уменьшено с 2.5
                # Интервалы атак немного уменьшаем, но не сильно
                self.basic_attack_interval = 1.0  # 1.2 -> 1.0
                self.circle_attack_interval = 8.0  # 10.0 -> 8.0
                self.sniper_attack_interval = 6.0  # 8.0 -> 6.0
            elif self.time_elapsed > CRITICAL_TIME:
                critical_factor = 1.0 + (self.time_elapsed - CRITICAL_TIME) / 40.0  # Медленнее рост
                self.current_difficulty = 1.8 * min(1.5, critical_factor)  # Меньший максимум
                # Интервалы атак остаются прежними
                self.basic_attack_interval = 1.0
                self.circle_attack_interval = 8.0
                self.sniper_attack_interval = 6.0
            else:
                self.current_difficulty = time_factor
                # Постепенное увеличение скорости, но не частоты
                self.basic_attack_interval = 1.2
                self.circle_attack_interval = 10.0
                self.sniper_attack_interval = 8.0

        def render(self, width, height, st, at):
            render = renpy.Render(width, height)

            current_time = renpy.get_game_runtime()
            self.time_elapsed = current_time - self.game_start_time

            if not self.game_over and self.time_elapsed >= GAME_DURATION:
                self.game_over = True
                self.game_over_time = current_time
                self.victory = True

            self.update_difficulty()

            # Обновляем помощника
            if not self.game_over:
                ability_used = self.assistant.update(
                    st - (self.assistant.old_st or 0),
                    current_time,
                    self.player.lives,
                    self.player.max_lives,
                    self.player.just_took_hit
                )

                if ability_used == 1:
                    if self.player.heal(1):
                        self.assistant_message = "Ты справишься!"
                        self.message_end_time = current_time + 2.0
                elif ability_used == 2:
                    self.player.activate_shield(3.0)
                    self.assistant_message = "Ты выдержишь!"
                    self.message_end_time = current_time + 2.0
                elif ability_used == 3:
                    self.player.activate_time_slow(3.0)
                    self.assistant_message = "Ты не проиграешь!"
                    self.message_end_time = current_time + 2.0

                self.player.reset_hit_flag()

            # Обновляем сообщение помощника
            if current_time >= self.message_end_time:
                self.assistant_message = ""

            # Добавляем атаки разных типов
            if not self.game_over:
                # Базовые атаки
                if current_time - self.last_basic_attack_time > self.basic_attack_interval:
                    self.add_basic_attack()
                    self.last_basic_attack_time = current_time

                # Атаки по кругу (реже)
                if current_time - self.last_circle_attack_time > self.circle_attack_interval:
                    self.add_circle_attack()
                    self.last_circle_attack_time = current_time

                # Снайперские атаки
                if current_time - self.last_sniper_attack_time > self.sniper_attack_interval:
                    self.add_sniper_attack()
                    self.last_sniper_attack_time = current_time

            if not self.game_over:
                self.check_collisions()

            # Применяем замедление времени к пулям
            if self.player.time_slow_active:
                for bullet in self.bullets:
                    if bullet.alive:
                        bullet.apply_time_slow(
                            self.player.x + self.player.size/2,
                            self.player.y + self.player.size/2,
                            self.player.time_slow_radius,
                            0.4  # Сильнее замедление
                        )

            # Рендерим помощника
            assistant_render = self.assistant.render(width, height, st, at)
            render.blit(assistant_render, (0, 0))

            # Рендерим игрока
            player_render = self.player.render(width, height, st, at)
            render.blit(player_render, (0, 0))

            # Рендерим пули
            bullets_to_remove = []
            for i, bullet in enumerate(self.bullets):
                if bullet.alive:
                    try:
                        bullet_render = bullet.render(width, height, st, at)
                        render.blit(bullet_render, (0, 0))
                    except Exception as e:
                        bullets_to_remove.append(i)
                else:
                    bullets_to_remove.append(i)

            for i in sorted(bullets_to_remove, reverse=True):
                if i < len(self.bullets):
                    self.bullets.pop(i)

            if self.game_over and self.game_over_time:
                if current_time - self.game_over_time > 1.0:
                    renpy.timeout(0)

            renpy.redraw(self, 0.01)
            return render

        def event(self, ev, x, y, st):
            if not self.game_over:
                result = self.player.event(ev, x, y, st)
                if result:
                    return result

            raise renpy.IgnoreEvent()
